cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(microchrony LANGUAGES CXX)
endif()
add_library(microchrony microchrony.cpp)
target_include_directories(microchrony PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(microchrony PUBLIC cxx_std_20)
set_target_properties(microchrony PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(microchrony PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

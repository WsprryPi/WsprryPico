cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "Target board")
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/pico_sdk_import.cmake")
project(Gp15LedParity LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.3.1" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "15.3.1")
    message(FATAL_ERROR "Use retained Pico SDK 2.3.1 and Arm GCC 15.3.1")
endif()
add_executable(Gp15LedParity main.cpp)
target_link_libraries(Gp15LedParity PRIVATE pico_stdlib)
pico_enable_stdio_uart(Gp15LedParity 0)
pico_enable_stdio_usb(Gp15LedParity 1)
pico_add_extra_outputs(Gp15LedParity)

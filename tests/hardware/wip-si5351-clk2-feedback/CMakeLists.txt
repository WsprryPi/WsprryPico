cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "Target board")
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/pico_sdk_import.cmake")
project(Si5351Clk2Feedback LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.3.1" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "15.3.1")
    message(FATAL_ERROR "Use retained Pico SDK 2.3.1 and Arm GCC 15.3.1")
endif()
add_executable(Si5351Clk2Feedback main.cpp)
target_link_libraries(Si5351Clk2Feedback PRIVATE pico_stdlib pico_unique_id hardware_i2c hardware_pwm)
pico_enable_stdio_uart(Si5351Clk2Feedback 0)
pico_enable_stdio_usb(Si5351Clk2Feedback 1)
pico_add_extra_outputs(Si5351Clk2Feedback)

if(NOT PICO_NO_FLASH)
    message(FATAL_ERROR "This diagnostic must run from RAM: use -DPICO_NO_FLASH=1")
endif()

cmake_minimum_required(VERSION 3.24)
project(HostnamePiCompatibility LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Run Pi C++ validation on identified Linux only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT EXPECTED_PICO MATCHES "^[0-9a-f]+$" OR NOT EXPECTED_PI MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Supply explicit immutable source identities")
endif()
foreach(pair IN ITEMS "PI_SOURCE|${EXPECTED_PI}" "PICO_SOURCE|${EXPECTED_PICO}" "MBEDTLS_SOURCE|0bebf8b8c7f07abe3571ded48a11aa907a1ffb20")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 var)
    list(GET parts 1 pin)
    string(LENGTH "${pin}" length)
    execute_process(COMMAND git -C "${${var}}" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND git -C "${${var}}" status --porcelain --untracked-files=normal OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT length EQUAL 40 OR NOT revision STREQUAL pin OR NOT dirty STREQUAL "")
        message(FATAL_ERROR "${var} must be clean at the exact supplied pin ${pin}")
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MBEDTLS_DIR "${MBEDTLS_SOURCE}")
set(MBEDTLS_CONFIG_FILE "${PICO_SOURCE}/src/network/pico/mbedtls_config.h")
set(GEN_FILES OFF)
set(USE_STATIC_MBEDTLS_LIBRARY ON)
set(USE_SHARED_MBEDTLS_LIBRARY OFF)
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON)
add_subdirectory("${MBEDTLS_SOURCE}/library" "${CMAKE_BINARY_DIR}/mbedtls")
# Compile the owned production core inventory, including its new hostname coordinator.
file(READ "${PICO_SOURCE}/CMakeLists.txt" pico_cmake)
string(REGEX MATCH "add_library\\(wsprrypico_core([^)]*)\\)" matched "${pico_cmake}")
separate_arguments(core UNIX_COMMAND "${CMAKE_MATCH_1}")
list(TRANSFORM core REPLACE "\\$\\{CMAKE_CURRENT_BINARY_DIR\\}/generated/web_assets.cpp" "${CMAKE_BINARY_DIR}/generated/web_assets.cpp")
foreach(source IN LISTS core)
    if(IS_ABSOLUTE "${source}")
        list(APPEND core_paths "${source}")
    else()
        list(APPEND core_paths "${PICO_SOURCE}/${source}")
    endif()
endforeach()
execute_process(COMMAND ${Python3_EXECUTABLE} "${PICO_SOURCE}/scripts/generate_web_assets.py" "${PICO_SOURCE}" "${CMAKE_BINARY_DIR}/generated/web_assets.cpp" COMMAND_ERROR_IS_FATAL ANY)
add_library(pico_fixture STATIC ${core_paths}
    "${PICO_SOURCE}/src/network/pico/consumer_radio_crypto.cpp"
    "${PICO_SOURCE}/src/network/pico/psa_lifetime.cpp")
target_include_directories(pico_fixture PUBLIC "${PICO_SOURCE}/src" "${PICO_SOURCE}/tests" "${MBEDTLS_SOURCE}/include")
target_compile_definitions(pico_fixture PUBLIC MBEDTLS_CONFIG_FILE="${MBEDTLS_CONFIG_FILE}" WSPRRY_PICO_STANDALONE_WSPR_BASE_FREQUENCY_HZ=3570100)
target_compile_options(pico_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(pico_fixture PUBLIC mbedcrypto)
set(client_sources codec.cpp frame_parser.cpp wire.cpp session.cpp detail/json.cpp)
list(TRANSFORM client_sources PREPEND "${PI_SOURCE}/src/WTP-Client/src/")
add_library(unchanged_client STATIC ${client_sources})
target_include_directories(unchanged_client PUBLIC "${PI_SOURCE}/src/WTP-Client/include")
file(READ "${PI_SOURCE}/src/tests/network/consumer_interop.cpp" fixture)
if(NOT fixture MATCHES "int main\\(int argc, char \\*\\*argv\\)")
    message(FATAL_ERROR "Unsupported original fixture entry point")
endif()
string(REPLACE "int main(int argc, char **argv) {" "int unchanged_pair_main(int argc, char **argv) {" fixture "${fixture}")
string(REGEX REPLACE "\\}[^}]*$" "  return 0;\n}\n" fixture "${fixture}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/unchanged_fixture.hpp" "${fixture}")
set(products wtp_integration/consumer_radio.cpp wtp_integration/configuration.cpp
    wtp_integration/browser_api.cpp wtp_integration/network_http.cpp
    wtp_integration/plain_tcp.cpp wtp_integration/tls.cpp wtp_runtime_bridge.cpp
    wtp_integration/output_lease.cpp wtp_integration/usb_cdc.cpp privileged_network_policy.cpp
    backend_http_guard.cpp support_request_guard.cpp httplib.cpp
    wtp_integration/application.cpp wtp_integration/status.cpp wtp_integration/scheduler.cpp
    wtp_integration/backend.cpp wtp_integration/execution_plan.cpp
    WSPR-Transmitter/src/execution_plan_compiler.cpp WSPR-Transmitter/src/transmission_controller.cpp)
list(TRANSFORM products PREPEND "${PI_SOURCE}/src/")

add_executable(hostname_pi_compatibility consumer_interop.cpp ${products})
target_compile_definitions(hostname_pi_compatibility PRIVATE WSPRRYPI_CONSUMER_CRYPTO_TEST=1 WSPRRY_PICO_MULTI_SESSIONS=1 WSPRRY_PICO_RECOVERY=1 WSPRRY_PICO_TYPE23=1)
target_compile_options(hostname_pi_compatibility PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_include_directories(hostname_pi_compatibility PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PI_SOURCE}/src" "${PI_SOURCE}/src/tests/network" "${PI_SOURCE}/src/WSPR-Transmitter/src")
target_link_libraries(hostname_pi_compatibility PRIVATE pico_fixture unchanged_client ssl crypto pthread)
